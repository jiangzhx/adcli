# 更新项目自选素材状态

# 请求地址

[https://api.oceanengine.com/open_api/v3.0/tools/project_material_type/update/](https://api.oceanengine.com/open_api/v3.0/tools/project_material_type/update/)[可视化调试](https://open.oceanengine.com/tools/visual_debug.html?docId=1869423189855306)[SDK下载](https://open.oceanengine.com/labels/7/docs/1773083358100557)

# 请求方法

POST

# Header

字段

类型

描述

Access-Token

string

Content-Type

string

# 请求参数

字段

类型

描述

advertiser_id

number

投放账户id

project_id

number

material_switch

string

注意：material_switch 与 original_video_title 至少传一个

original_video_title

string

注意：material_switch 与 original_video_title 至少传一个，且 original_video_title 当且仅当星广联投项目时支持传入修改

material_list

object

video_material_list

object[]

image_mode

string

video_id

string

视频ID，取值来源【[获取视频素材](https://open.oceanengine.com/labels/8/docs/1696710601820172)】

video_cover_id

string

视频封面图片ID，可通过[【获取视频素材】](https://open.oceanengine.com/labels/7/docs/1696710601820172?origin=left_nav)[【获取图片素材】](https://open.oceanengine.com/labels/7/docs/1696710601254912)[【获取视频智能封面】](https://open.oceanengine.com/labels/7/docs/1696710602404864?origin=left_nav)接口获取。

item_id

number

video_hp_visibility

string

guide_video_id

string

image_material_list

object[]

image_mode

string

image_id

string

图片ID，取值来源【[获取图片素材](https://open.oceanengine.com/labels/7/docs/1696710601254912?origin=left_nav)】

carousel_material_list

object[]

carousel_id

number

图文id，取值来源【[获取图文素材](https://open.oceanengine.com/labels/7/docs/1773554026740736)】

item_id

number

抖音图文id，取值来源【[获取创编可用的抖音图文素材](https://open.oceanengine.com/docs/1812353010729227)】

video_hp_visibility

string

trial_play_material_list

object[]

app_play_uri

string

试玩素材uri，取值来源【[获取试玩/直玩素材列表](https://open.oceanengine.com/docs/1836615534374475)】

guide_video_id

string

instant_play_material_list

object[]

直玩素材信息

app_play_uri

string

直玩素材uri，取值来源【[获取试玩/直玩素材列表](https://open.oceanengine.com/docs/1836615534374475)】

title_material_list

object[]

title

string

product_info

object

titles

string[]

image_ids

string[]

产品主图，取值来源【[获取图片素材](https://open.oceanengine.com/docs/1696710601254912)】，上限10

selling_points

string[]

call_to_action_buttons

string[]

source

string

来源，字数[2-10]

anchor_related_type

string

anchor_material_list

object[]

原生锚点素材，anchor_related_type=SELECT时必填，数量上限1

anchor_type

string

anchor_id

number

原生锚点id，取值来源【[获取账户下的原生锚点](https://open.oceanengine.com/labels/7/docs/1757715831323652?origin=left_nav)】，anchor_related_type=SELECT时必填

component_material_list

object[]

component_id

number

组件id，取值来源【[查询组件列表](https://open.oceanengine.com/docs/1696710673645580)】，支持横版/竖版视频

external_url_material_list

string[]

web_url_material_list

string[]

playlet_series_url_list

string[]

短剧合集链接url
